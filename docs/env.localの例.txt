# .env.local の書き方の例
# 実際の値は Supabase の API Keys からコピーして貼り付けてください。

# ダブルクォートで囲む書き方（推奨）
NEXT_PUBLIC_SUPABASE_URL="https://xxxxxxxx.supabase.co"
NEXT_PUBLIC_SUPABASE_ANON_KEY="eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.xxxxxxxx..."

# 囲まない書き方でも動く（値の前後にスペースを入れないこと）
# NEXT_PUBLIC_SUPABASE_URL=https://xxxxxxxx.supabase.co
# NEXT_PUBLIC_SUPABASE_ANON_KEY=eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.xxxxxxxx...

# 注意:
# - 使うファイルは .env.local です（.env.example は雛形なので本番のキーを書かないこと）
# - = の前後にスペースを入れない
# - キーは「anon public」のもの。「service_role secret」は使わない
# - キーの途中で改行しない（1行で収める）
